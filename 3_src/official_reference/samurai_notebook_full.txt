def resolve_training_archive(sequence, modality):
    sensor, target = sequence.split("-", 1)
    folder = (TRAIN_HSI_FOLDERS if modality == "hsi" else TRAIN_COLOR_FOLDERS)[sensor]

    sensor, target = sequence.split("-", 1)
    if modality == "color":

def direct_drive_url(remote_path):
    file_id = parse_qs(urlparse(url_by_path[remote_path]).query)["id"][0]
    return f"https://drive.usercontent.google.com/download?id={file_id}&export=download&confirm=t"

def download_drive_file(remote_path, local_path):
    local_path = Path(local_path)
    local_path.parent.mkdir(parents=True, exist_ok=True)
    if local_path.exists():
        return local_path

    temporary_path = local_path.with_suffix(local_path.suffix + ".part")
    with requests.get(direct_drive_url(remote_path), stream=True, timeout=(15, 180)) as response:
        response.raise_for_status()
        if "text/html" in response.headers.get("content-type", ""):
            raise RuntimeError(f"Drive returned an error page for {remote_path}")
        with temporary_path.open("wb") as file:
            for chunk in response.iter_content(chunk_size=4 * 1024**2):
                if chunk:
                    file.write(chunk)

    temporary_path.replace(local_path)
    return local_path

def download_many(jobs, workers=16, description="Downloading"):
    records = jobs[["remote_path", "local_path"]].to_dict("records")
    if not records:
        return pd.DataFrame(columns=["remote_path", "error"])

    errors = []
    with ThreadPoolExecutor(max_workers=min(workers, len(records))) as pool:
        futures = {pool.submit(download_drive_file, job["remote_path"], job["local_path"]): job for job in records}
        for future in tqdm(as_completed(futures), total=len(futures), desc=description):
            job = futures[future]
            try:
                future.result()
            except Exception as error:
                errors.append({"remote_path": job["remote_path"], "error": str(error).splitlines()[0]})

    print(f"Completed: {len(records) - len(errors):,}/{len(records):,}")
    return pd.DataFrame(errors)

MOSAIC_PATTERNS = {"nir": (5, 5), "rednir": (4, 4), "vis": (4, 4)}

def mosaic_to_cube(image, sensor):
    pattern_h, pattern_w = MOSAIC_PATTERNS[sensor]
    height = image.shape[0] // pattern_h * pattern_h
    width = image.shape[1] // pattern_w * pattern_w
    image = image[:height, :width]
    return image.reshape(height // pattern_h, pattern_h, width // pattern_w, pattern_w).transpose(0, 2, 1, 3).reshape(height // pattern_h, width // pattern_w, pattern_h * pattern_w)

def target_context_profiles(cube, box):
    height, width, bands = cube.shape
    x, y, box_width, box_height = box
    x1 = int(np.clip(round(x), 0, width - 1)); y1 = int(np.clip(round(y), 0, height - 1))
    x2 = int(np.clip(round(x + box_width), x1 + 1, width)); y2 = int(np.clip(round(y + box_height), y1 + 1, height))
    margin_x = max(int(round(box_width)), 4); margin_y = max(int(round(box_height)), 4)
    context_x1 = max(0, x1 - margin_x); context_y1 = max(0, y1 - margin_y)
    context_x2 = min(width, x2 + margin_x); context_y2 = min(height, y2 + margin_y)

    target_pixels = cube[y1:y2, x1:x2].reshape(-1, bands)
    context_crop = cube[context_y1:context_y2, context_x1:context_x2]
    context_mask = np.ones(context_crop.shape[:2], dtype=bool)
    context_mask[y1 - context_y1:y2 - context_y1, x1 - context_x1:x2 - context_x1] = False
    context_pixels = context_crop[context_mask]

    target_profile = target_pixels.mean(axis=0)
    context_profile = context_pixels.mean(axis=0)
    target_profile /= target_profile.mean() + 1e-6
    context_profile /= context_profile.mean() + 1e-6
    return target_profile, context_profile

import argparse, gc, json, time, warnings
from pathlib import Path
import numpy as np, torch

warnings.filterwarnings("ignore", message="Flash Attention is disabled.*")
from sam2.build_sam import build_sam2_video_predictor
import sam2.sam2_video_predictor as sam2_video_module
import sam2.utils.misc as sam2_misc

def no_progress(iterable, *args, **kwargs):
    return iterable

sam2_video_module.tqdm = no_progress
sam2_misc.tqdm = no_progress

def load_initial_box(path):
    text = Path(path).read_text().replace(",", " ").replace("\t", " ")
    values = np.fromstring(text, sep=" ", dtype=float)
    if len(values) < 4:
        raise ValueError(f"Could not read an x, y, width, height box from {path}")
    return values[:4]

def valid_result(path, expected_frames):
    try:
        return np.load(path).shape == (expected_frames, 4)
    except Exception:
        return False

def mask_to_box(mask, previous_box):
    mask = mask.detach().cpu().numpy() > 0
    y, x = np.where(mask)
    if len(x) == 0:
        return previous_box.copy(), True
    return np.array([x.min(), y.min(), x.max() - x.min() + 1, y.max() - y.min() + 1], dtype=float), False

def track_sequence(predictor, task):
    frame_dir, result_path = Path(task["frame_dir"]), Path(task["result_path"])
    expected_frames = int(task["frames"])
    if valid_result(result_path, expected_frames):
        print(f'{task["sequence"]}: already complete', flush=True)
        return

    frame_count = len(list(frame_dir.glob("*.jpg")))
    if frame_count != expected_frames:
        raise RuntimeError(f'{task["sequence"]}: found {frame_count} frames, expected {expected_frames}')

    initial_box = load_initial_box(task["init_path"])
    x, y, width, height = initial_box
    prompt_box = np.array([x, y, x + width, y + height], dtype=np.float32)
    predictions = np.full((expected_frames, 4), np.nan, dtype=float)
    previous_box, empty_masks = initial_box.copy(), 0
    start = time.perf_counter()

    state = predictor.init_state(str(frame_dir), offload_video_to_cpu=True, offload_state_to_cpu=True, async_loading_frames=True)
    predictor.add_new_points_or_box(state, frame_idx=0, obj_id=0, box=prompt_box)
    for frame_index, _, masks in predictor.propagate_in_video(state):
        box, empty = mask_to_box(masks[0, 0], previous_box)
        if frame_index < expected_frames:
            predictions[frame_index] = box
        previous_box = box
        empty_masks += int(empty)

    predictions[0] = initial_box
    for frame_index in range(1, expected_frames):
        if not np.isfinite(predictions[frame_index]).all():
            predictions[frame_index] = predictions[frame_index - 1]

    result_path.parent.mkdir(parents=True, exist_ok=True)
    temporary_path = result_path.with_suffix(".npy.part")
    with temporary_path.open("wb") as file:
        np.save(file, predictions)
    temporary_path.replace(result_path)

    predictor.reset_state(state)
    del state
    gc.collect(); torch.cuda.empty_cache()
    seconds = time.perf_counter() - start
    print(f'{task["sequence"]}: {expected_frames} frames, {empty_masks} empty masks, {seconds:.1f}s', flush=True)

def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--tasks", required=True)
    parser.add_argument("--checkpoint", required=True)
    parser.add_argument("--config", required=True)
    args = parser.parse_args()

    tasks = json.loads(Path(args.tasks).read_text())
    pending = [task for task in tasks if not valid_result(task["result_path"], int(task["frames"]))]
    if not pending:
        return

    torch.cuda.set_device(0)
    predictor = build_sam2_video_predictor(args.config, args.checkpoint, device="cuda:0")
    print(f"Model ready on {torch.cuda.get_device_name(0)}", flush=True)
    with torch.inference_mode(), torch.autocast("cuda", dtype=torch.float16):
        for task in pending:
            track_sequence(predictor, task)

if __name__ == "__main__":
    main()

    final_predictions = pd.read_csv(prediction_path)
    box_columns = ["x", "y", "width", "height"]
    missing_ids = set(submission_template["ID"]) - set(final_predictions["ID"])
    extra_ids = set(final_predictions["ID"]) - set(submission_template["ID"])
    problems = []

    if final_predictions["ID"].duplicated().any():
        problems.append("duplicated prediction IDs")
    if missing_ids:
        problems.append(f"{len(missing_ids)} missing IDs")
    if extra_ids:
        problems.append(f"{len(extra_ids)} unexpected IDs")
    if problems:
        raise RuntimeError("Submission validation failed: " + ", ".join(problems))

    submission = submission_template[["ID"]].merge(final_predictions, on="ID", how="left", sort=False)
    submission = submission[submission_template.columns]

    if submission[box_columns].isna().any().any():
        raise RuntimeError("The submission contains missing bounding boxes")

    submission[box_columns] = np.rint(submission[box_columns]).astype(int)
    if (submission[["x", "y"]] < 0).any().any() or (submission[["width", "height"]] <= 0).any().any():
        raise RuntimeError("The submission contains invalid bounding boxes")
    if not submission["ID"].equals(submission_template["ID"]):
        raise RuntimeError("Submission IDs are not in the required order")

    submission.to_csv(SUBMISSION_PATH, index=False)
    saved_submission = pd.read_csv(SUBMISSION_PATH)
    if len(saved_submission) != len(submission_template) or list(saved_submission.columns) != list(submission_template.columns):
        raise RuntimeError("The saved submission has an incorrect shape or columns")

    display(pd.DataFrame([{
        "rows": len(submission),
        "sequences": test["sequence"].nunique(),
        "missing_values": int(submission.isna().sum().sum()),
        "file_mb": SUBMISSION_PATH.stat().st_size / 1024**2
    }]).round(3))
    display(pd.concat([submission.head(3), submission.tail(3)]))
    print(f"Submission ready: {SUBMISSION_PATH}")
